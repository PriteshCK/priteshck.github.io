sns.set_theme(style="whitegrid")
palette = sns.color_palette()

# Set up the figure and axis
fig, ax1 = plt.subplots(figsize=(12, 5))

# Plot temperature anomaly (left y-axis)
sns.lineplot(data=data, x="Year", y="Temp_Anomaly", ax=ax1, color=palette[3], 
             marker="o", linewidth=2, label="Temperature Anomaly")

ax1.tick_params(axis="y", labelcolor=palette[3])

plt.title("Global Temperature", fontsize=16, fontweight="bold")

# Legends (positioned to avoid overlap)
ax1.legend(loc="upper left", bbox_to_anchor=(0, 1), frameon=True)

# Grid customization
ax1.grid(True, linestyle="--", alpha=0.6)

# Show the plot
plt.show()
